# PyInstaller version resource for trustnode-service.exe.
#
# Without this the service ships with EMPTY version info, so Task Manager and
# Details show a blank publisher and no description - and next to an Electron
# shell that also had no branding, an operator sees "Electron" and a nameless
# helper instead of TrustNode. Branding is not decoration here: when an
# operator is asked "what is using your memory?", the answer has to name us.
VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(0, 1, 0, 0),
    prodvers=(0, 1, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo([
      StringTable(
        '040904B0',
        [StringStruct('CompanyName', 'TrustNode'),
         StringStruct('FileDescription', 'TrustNode Edge Service'),
         StringStruct('FileVersion', '0.1.0.0'),
         StringStruct('InternalName', 'trustnode-service'),
         StringStruct('LegalCopyright', 'TrustNode'),
         StringStruct('OriginalFilename', 'trustnode-service.exe'),
         StringStruct('ProductName', 'TrustNode Edge'),
         StringStruct('ProductVersion', '0.1.0.0')])
    ]),
    VarFileInfo([VarStruct('Translation', [1033, 1200])])
  ]
)
